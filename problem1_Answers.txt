INSERT INTO EMPLOYEE (ssn, fname, lname, bdate, address, gender, salary, Dno) VALUES (99, 'Test', 'Test', '1999-09-09', 'Test', 'T', 9999.99, NULL);
INSERT INTO DEPARTMENT (Dnumber, Dname, mgr_ssn, mgr_start_date) VALUES (999, 'Test', 99, '2000-01-01');
DELETE FROM EMPLOYEE WHERE ssn = 99;
